  reader.onload = () => {
    let parsed;
    try{ parsed = JSON.parse(reader.result); }
    catch(err){ alert('That file isn\'t valid JSON — couldn\'t read it.'); return; }

    const incoming = Array.isArray(parsed) ? parsed : parsed.wraps;
    if(!Array.isArray(incoming)){
      alert('This doesn\'t look like a Wrappd backup file.');
      return;
    }

    loadWraps();
    const existingIds = new Set(wraps.map(w=>w.id));
    const newCount = incoming.filter(w => !existingIds.has(w.id)).length;
    const overlapCount = incoming.length - newCount;

    let msg = `Found ${incoming.length} wrap${incoming.length===1?'':'s'} in this file`;
    if(overlapCount>0) msg += ` (${newCount} new, ${overlapCount} matching wraps already on this device)`;
    msg += '.\n\nClick OK to MERGE into your saved wraps (matching ones get overwritten with the file\'s version).\nClick Cancel to REPLACE your saved wraps entirely with just this file\'s wraps.';

    const merge = confirm(msg);
    if(merge){
      const byId = new Map(wraps.map(w=>[w.id,w]));
      incoming.forEach(w => byId.set(w.id, w));
      wraps = Array.from(byId.values());
    } else {

function computeAutoSub(w){
  const counts = { text:0, audio:0, both:0 };
  (w.books||[]).forEach(b => { if(counts[b.format]!==undefined) counts[b.format]++; });
  const parts = [];
  if(counts.text) parts.push(`${counts.text} ${counts.text===1?'book':'books'}`);
  if(counts.audio) parts.push(`${counts.audio} ${counts.audio===1?'audiobook':'audiobooks'}`);
  if(counts.both) parts.push(`${counts.both} mixed both`);
  return parts.length ? parts.join(', ') : '0 books';
}

  if(bg.type==='gradient') return `linear-gradient(${bg.angle||135}deg, ${bg.colors[0]}, ${bg.colors[1]})`;

/* Shrinks the fixed-size (PC/1080-wide) canvas visually to fit narrow phone
   screens using a CSS transform, so the design itself never changes — only
   how small it's shown. The underlying #canvas element always keeps its
   true dimensions, which is what export captures. */
function fitCanvasToViewport(){
  const outer = document.getElementById('canvasScaleOuter');
  const shell = document.getElementById('canvasShell');
  if(!outer || !shell) return;
  // measure natural (unscaled) size
  shell.style.transform = 'none';
  outer.style.width = '';
  outer.style.height = '';
  const naturalW = shell.offsetWidth;
  const naturalH = shell.offsetHeight;
  if(!naturalW || !naturalH) return;
  const available = outer.parentElement ? outer.parentElement.clientWidth : naturalW;
  const scale = Math.min(1, available / naturalW);
  if(scale < 1){
    shell.style.transform = `scale(${scale})`;
    outer.style.width = (naturalW*scale) + 'px';
    outer.style.height = (naturalH*scale) + 'px';
  }
}

function bindStickerEvents(el, st){
  el.addEventListener('pointerdown', (e) => {
    if(e.target.classList.contains('h-resize') || e.target.classList.contains('h-rotate') || e.target.classList.contains('h-del')) return;
    e.stopPropagation(); e.preventDefault();
    selectSticker(st.id);
    el.setPointerCapture(e.pointerId);
    const startX = e.clientX, startY = e.clientY;
    const ox = st.x, oy = st.y;
    let raf = null, pending = null;
    function apply(){
      raf = null;
      st.x = pending.x; st.y = pending.y;
      el.style.transform = stickerTransform(st);
    }
    function move(ev){
      pending = { x: ox + (ev.clientX - startX), y: oy + (ev.clientY - startY) };
      if(raf===null) raf = requestAnimationFrame(apply);
    }
    function up(){
      if(raf) cancelAnimationFrame(raf);
      if(pending){ st.x = pending.x; st.y = pending.y; el.style.transform = stickerTransform(st); }
      el.releasePointerCapture(e.pointerId);
      el.removeEventListener('pointermove', move);
      el.removeEventListener('pointerup', up);
    }
    el.addEventListener('pointermove', move);
    el.addEventListener('pointerup', up);
  });
  const delH = el.querySelector('.h-del');
  delH.addEventListener('pointerdown', e => e.stopPropagation());
  delH.addEventListener('click', (e) => {
    e.stopPropagation();
    wrap.stickers = wrap.stickers.filter(s=>s.id!==st.id);
    selectedStickerId = null;
    renderStickers();
  });
  const resizeH = el.querySelector('.h-resize');
  resizeH.addEventListener('pointerdown', (e) => {
    e.stopPropagation(); e.preventDefault();
    selectSticker(st.id);
    resizeH.setPointerCapture(e.pointerId);
    const startX = e.clientX; const ow = st.w;
    let raf=null, pendingW=null;
    function apply(){
      raf=null; st.w = pendingW; st.h = pendingW;
      el.style.width = st.w+'px'; el.style.height = st.h+'px';
      if(st.type==='emoji') el.style.fontSize = (st.w*0.8)+'px';
    }
    function move(ev){
      const delta = ev.clientX - startX;
      pendingW = Math.max(20, ow + delta);
      if(raf===null) raf = requestAnimationFrame(apply);
    }
    function up(){
      if(raf) cancelAnimationFrame(raf);
      if(pendingW!==null) apply();
      resizeH.releasePointerCapture(e.pointerId);
      resizeH.removeEventListener('pointermove', move);
      resizeH.removeEventListener('pointerup', up);
    }
    resizeH.addEventListener('pointermove', move);
    resizeH.addEventListener('pointerup', up);
  });
  const rotH = el.querySelector('.h-rotate');
  rotH.addEventListener('pointerdown', (e) => {
    e.stopPropagation(); e.preventDefault();
    selectSticker(st.id);
    rotH.setPointerCapture(e.pointerId);
    const rect = el.getBoundingClientRect();
    const cx = rect.left + rect.width/2, cy = rect.top + rect.height/2;
    let raf=null, pendingRot=null;
    function apply(){ raf=null; st.rot = pendingRot; el.style.transform = stickerTransform(st); }
    function move(ev){
      pendingRot = Math.round(Math.atan2(ev.clientY-cy, ev.clientX-cx) * 180/Math.PI + 90);
      if(raf===null) raf = requestAnimationFrame(apply);
    }
    function up(){
      if(raf) cancelAnimationFrame(raf);
      if(pendingRot!==null) apply();
      rotH.releasePointerCapture(e.pointerId);
      rotH.removeEventListener('pointermove', move);
      rotH.removeEventListener('pointerup', up);
    }
    rotH.addEventListener('pointermove', move);
    rotH.addEventListener('pointerup', up);
  });
}

    outer.style.width = prevW; outer.style.height = prevH;
  }
  return { dataUrl: shot.toDataURL('image/png') };
}

document.getElementById('exportBtn').onclick = async () => {
  const btn = document.getElementById('exportBtn');
  const old = btn.textContent; btn.textContent = 'Rendering…'; btn.disabled = true;
  try{
    const { dataUrl } = await renderWrapToDataUrl();
    if(isIOS()){
      showSaveOverlay(dataUrl);
    } else {
      const link = document.createElement('a');
      link.download = (wrap.name || 'reading-wrap').replace(/\s+/g,'-').toLowerCase() + '.png';
      link.href = dataUrl;
      link.click();
    }
  }catch(err){
    alert('Export ran into an issue (likely a cross-origin image). Try re-uploading any tricky covers/backgrounds as your own images, then export again.');
    console.error(err);
  }
  btn.textContent = old; btn.disabled = false;
};
